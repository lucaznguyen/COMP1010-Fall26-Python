\section*{Problem 1: Break Down the Time}

You are given a number of seconds since midnight. Convert it to hours, minutes, and seconds. For example, 3,661 seconds is 1 hour, 1 minute, and 1 second.

\subsection*{Input}
\begin{itemize}
    \item One integer \texttt{totalSeconds}, where $0 \leq \texttt{totalSeconds} < 86400$.
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print the hours, minutes, and remaining seconds on one line, separated by spaces.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
3661

Output:
1 1 1
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
59

Output:
0 0 59
\end{verbatim}
\end{multicols}
\end{example*}

\section*{Problem 2: Can You Borrow a Library Book?}

A person can borrow a book if they have a valid library card and do not have any overdue books. Given the two yes/no answers, report whether the card is valid, whether there are no overdue books, and whether the person can borrow a book. Each answer is the exact word \texttt{True} or \texttt{False}.

\subsection*{Input}
\begin{itemize}
    \item Two lines: whether the person has a valid library card, then whether they have overdue books.
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print \texttt{Card is valid: ...}, \texttt{No overdue books: ...}, and \texttt{Can borrow: ...}, each on its own line. Replace \texttt{...} with \texttt{True} or \texttt{False}.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
True
False

Output:
Card is valid: True
No overdue books: True
Can borrow: True
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
True
True

Output:
Card is valid: True
No overdue books: False
Can borrow: False
\end{verbatim}
\end{multicols}
\end{example*}

\section*{Problem 4: Create a Profile Line}

Given a first name, a last name, and a course number, join them into one profile line in the format shown. The hyphen is the literal dash character \texttt{-} between the name and the word \texttt{Course}; print it as text, not as a subtraction sign.

\subsection*{Input}
\begin{itemize}
    \item Three lines: a first name, a last name (each one word), and an integer course number ($1 \leq \texttt{courseNumber} \leq 9999$).
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print one line in this format: \texttt{Profile: firstName lastName - Course courseNumber}.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
Mia
Tran
101

Output:
Profile: Mia Tran - Course 101
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
An
Le
3

Output:
Profile: An Le - Course 3
\end{verbatim}
\end{multicols}
\end{example*}
