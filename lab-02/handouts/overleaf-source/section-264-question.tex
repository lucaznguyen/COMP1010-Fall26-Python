\section*{Problem 1: Power Calculation}

Given two non-negative integers \texttt{a} and \texttt{b}, calculate their sum, product, and \texttt{a} raised to the power of \texttt{b}.

\subsection*{Input}
\begin{itemize}
    \item One line containing \texttt{a} and \texttt{b}, separated by a space, where $0 \leq a \leq 10$ and $0 \leq b \leq 6$.
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print \texttt{a + b}, \texttt{a * b}, and \texttt{a ** b}, separated by spaces.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
2 3

Output:
5 6 8
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
5 0

Output:
5 0 1
\end{verbatim}
\end{multicols}
\end{example*}

\section*{Problem 2: Can You Enter the Campus Event?}

A campus event is open to people aged 18 or older who have an invitation or a student card. Given the person's age and the two yes/no answers, report whether they meet the age rule, whether they have an invitation or student card, and whether they can enter. Each yes/no answer is the exact word \texttt{True} or \texttt{False}.

\subsection*{Input}
\begin{itemize}
    \item Three lines: the person's age, whether they have an invitation, and whether they have a student card. The age is between 0 and 120.
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print \texttt{Age requirement: ...}, \texttt{Invitation/student-card requirement: ...}, and \texttt{Can enter: ...}, each on its own line. Replace \texttt{...} with \texttt{True} or \texttt{False}.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
19
False
True

Output:
Age requirement: True
Invitation/student-card requirement: True
Can enter: True
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
16
True
False

Output:
Age requirement: False
Invitation/student-card requirement: True
Can enter: False
\end{verbatim}
\end{multicols}
\end{example*}

\section*{Problem 3: Letter from Its Number}

A number from 65 to 90 represents an uppercase English letter. Given such a number, print the letter it represents.

\subsection*{Input}
\begin{itemize}
    \item One integer \texttt{code}, where $65 \leq \texttt{code} \leq 90$.
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print the uppercase letter represented by the number.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
65

Output:
A
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
90

Output:
Z
\end{verbatim}
\end{multicols}
\end{example*}

\section*{Problem 4: Make a Room Banner}

Given a one-word slogan, a repeat count, and a room number, repeat the slogan and print the room number next to it. The hyphen is the literal dash character \texttt{-} between the repeated slogan and the room information; print it as text, not as a subtraction sign.

\subsection*{Input}
\begin{itemize}
    \item Three lines: a one-word slogan (1 to 20 English letters), a repeat count from 1 to 5, and a room number ($1 \leq \texttt{roomNumber} \leq 999$).
\end{itemize}

\subsection*{Output}
\begin{itemize}
    \item Print one line in this format: \texttt{Banner: repeatedSlogan - Room roomNumber}.
\end{itemize}

\vspace{0.2cm}

\begin{example*}{}{}
\begin{multicols}{2}
\textbf{Example 1}
\begin{verbatim}
Input:
Go
3
204

Output:
Banner: GoGoGo - Room 204
\end{verbatim}
\textbf{Example 2}
\begin{verbatim}
Input:
Hi
1
7

Output:
Banner: Hi - Room 7
\end{verbatim}
\end{multicols}
\end{example*}
